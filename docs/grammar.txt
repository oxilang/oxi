Id -> x | y | z | ...
Lit -> 0 | 1 | -1 | ...
     | 0.1 | -0.1 | ...
     | "a" | "ab" | ...
     | 'a' | 'b' | ...
     | true | false
Path -> Id [::< [Ty],* >]? [::Id [::< [Ty],* >]?]*
ITree -> Path [as Id]?
       | Path::{[ITree],+}
       | Path::*
Ty -> Path
    | & [mut]? Ty
    | [Ty]
    | [Ty; 1] | [Ty; 2] | ...
    | ([Ty],*) -> Ty
    | (Ty, Ty, [Ty],*)
    | !
Expr -> Lit
      | Path
      | Expr + Expr | Expr - Expr | Expr * Expr | Expr / Expr | Expr % Expr
      | Expr << Expr | Expr >> Expr
      | Expr & Expr | Expr [|] Expr | Expr ^ Expr
      | Expr == Expr | Expr != Expr | Expr > Expr | Expr < Expr | Expr <= Expr | Expr >= Expr
      | Expr && Expr | Expr || Expr
      | Expr .. Expr
      | Expr |> Expr
      | Expr@ | Expr?
      | @Expr | !Expr | -Expr | & [mut]? Expr
      | Expr = Expr | Expr += Expr | Expr -= Expr | Expr *= Expr | Expr /= Expr | Expr %= Expr
      | Path { [Id: Expr],* }
      | [ [Expr],* ]
      | Expr([Expr],*)
      | Expr.Id
      | Expr as Ty
      | (Expr, Expr, [Expr],*)
      | { [Stmt]* [Expr]? }
      | (Expr)
      | break [Expr]?
      | return [Expr]?
      | if Expr { [Stmt]* [Expr]? } [else Expr]?
      | while Expr { [Stmt]* }
      | loop { [Stmt]* }
Stmt -> Expr;
      | let [mut]? Id [: Ty]? [= Expr]?;
Assoc -> [pub]? type Id [= Ty]?;
       | [pub]? fn Id [< [Id],* >]? ([Id: Ty],*) Ty { [Stmt]* [Expr]? }
Item -> [pub]? const Id: Ty = Expr;
      | [pub]? type Id [< [Id],* >]? = Ty;
      | [pub]? struct Id [< [Id],* >]? { [[pub]? Id: Ty],* [Assoc]* }
      | [pub]? trait Id [< [Id],* >]? { [Assoc]* }
      | impl Path for Path { [Assoc]* }
      | [pub]? [extern]? fn Id [< [Id],* >]? ([Id: Ty],*) Ty { [Stmt]* [Expr]? }
      | [pub]? [extern]? fn Id [< [Id],* >]? ([Id: Ty],*) Ty;
      | [pub]? import ITree;
      | [pub]? mod Id [{ [Item]* }]?;
Program -> [Item]*
